# The script:
    import json, subprocess, sys, time, urllib.request
    M = "/Users/andy/devel/fornace/models/qwen2.5-0.5b-instruct"
    prompt = open("/Users/andy/devel/fornace/bench/prompts/alice-512.txt").read()
    def run(fmt, mode):
        cmd = ["llama-server", "-m", f"{M}/model-{fmt}.llama.gguf", "-ngl", "99", "-fa", "on", "-lm", mode,
               "-c", "4096", "-np", "1", "--no-cache-prompt", "-cram", "0", "--host", "127.0.0.1", "--port", "8096"]
        p = subprocess.Popen(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        for _ in range(240):
            try: urllib.request.urlopen("http://127.0.0.1:8096/health"); break
            except Exception: time.sleep(0.25)
        body = json.dumps({"prompt": prompt, "max_tokens": 128, "stream": True, "temperature": 0,
                           "ignore_eos": True, "repeat_penalty": 1.0, "cache_prompt": False}).encode()
        out = []
        for i in range(4):
            req = urllib.request.Request("http://127.0.0.1:8096/v1/completions", body, {"Content-Type": "application/json"})
            first = last = None
            with urllib.request.urlopen(req) as r:
                for line in r:
                    if line.startswith(b"data: {") and json.loads(line[6:])["choices"][0]["text"]:
                        last = time.perf_counter(); first = first or last
            if i: out.append(127 / (last - first))
        p.terminate(); p.wait()
        return sum(out) / len(out)
    for fmt in ("q4_0", "bf16"):
        res = {m: [] for m in ("mmap", "none")}
        for mode in ("mmap", "none", "none", "mmap"):
            res[mode].append(run(fmt, mode))
        print(fmt, {m: [round(v, 1) for v in vs] for m, vs in res.items()},
              "none/mmap %.3f" % (sum(res["none"]) / sum(res["mmap"])))
